  9 fn gen_color_coef() -> f32 {
 10     let mut rng = rand::thread_rng();
 11     rng.gen_range(1.4, 8.0)
 12 }

 14 struct Settings {
 15     r: f32,
 16     g: f32,
 17     b: f32,
 18     max_iter: i32,
 19     colors: Vec<Color>,
 20 }
 21 
 22 fn mandelbrot(x: f64, y: f64, set: &Settings) -> Color {
 23     let mut zx = 0f64;
 24     let mut zy = 0f64;
 25 
 26     let mut i = 0;
 27 
 28     while i < set.max_iter && (zx * zx + zy * zy) <= 4f64 {
 29         let zx_temp = zx * zx - zy * zy + x;
 30         zy = 2f64 * zx * zy + y;
 31         zx = zx_temp;
 32         i += 1;
 33     }
 34 
 35     if i == set.max_iter {
 36         return BLACK;
 37     } else {
 38 
 39         return set.colors[i as usize];
 40         // let mut i = i as f32;
 41         // if i > 255. {
 42         //     i -= 255.
 43         // }
 44         // return Color::new(set.r * i / 255., set.g * i / 255., set.b * i / 255., 1.0);
 45     }
 46 }

 48 fn mandelbrot_image(x_0: f32, y_0: f32, zoom: f32, set: &Settings) -> Image {
 49     let mut screen = Image::gen_image_color(screen_width() as u16, screen_height() as u16, WHITE);
 50 
 51     let ox = screen_width() / 2.0;
 52     let oy = screen_height() / 2.0;
 53 
 54     let unit;
 55     if ox < oy {
 56         unit = ((zoom * 2.) / screen_width()).abs();
 57     } else {
 58         unit = ((zoom * 2.) / screen_height()).abs();
 59     };
 60 
 61     for xx in 0..screen_width() as u32 {
 62         for yy in 0..screen_height() as u32 {
 63             let x = (xx as f64 - ox as f64) * unit as f64 + x_0 as f64;
 64             let y = (yy as f64 - oy as f64) * unit as f64 + y_0 as f64;
 65             screen.set_pixel(xx, yy, mandelbrot(x, y, set));
 66         }
 67     }
 68     return screen;
 69 }

 71 #[macroquad::main("Texture")]
 72 async fn main() {
 73     let mut set: Settings = Settings {
 74         r: gen_color_coef(),
 75         g: gen_color_coef(),
 76         b: gen_color_coef(),
 77         max_iter: 55,
 78         colors: Vec::<Color>::new(),
 79     };
 80 
 81 
 82     let mut rng = rand::thread_rng();
 83     for i in 0..1000 {
 84         set.colors.push(Color::new(
 85             rng.gen_range(25., 255.) / 255.,
 86             rng.gen_range(25., 255.) / 255.,
 87             rng.gen_range(25., 255.) / 255.,
 88             1.0,
 89         ));
 90     }
 91 
 92     let mut rng = rand::thread_rng();
 93     let mut x_0 = 0.;
 94     let mut y_0 = 0.;
 95     let mut zoom = 2.0; // lower increase zoom
 96     let zoom_mult = 0.4; // lower increase zoom
 97     let dist = 0.2;
 98     let mut screen = mandelbrot_image(x_0, y_0, zoom, &set);
 99 
100     loop {
101         clear_background(LIGHTGRAY);
102 
103         if is_key_down(KeyCode::D) {
104             x_0 += dist * zoom;
105             screen = mandelbrot_image(x_0, y_0, zoom, &set);
106         }
107         if is_key_down(KeyCode::A) {
108             x_0 -= dist * zoom;
109             screen = mandelbrot_image(x_0, y_0, zoom, &set);
110         }
111         if is_key_down(KeyCode::S) {
112             y_0 += dist * zoom;
113             screen = mandelbrot_image(x_0, y_0, zoom, &set);
114         }
115         if is_key_down(KeyCode::W) {
116             y_0 -= dist * zoom;
117             screen = mandelbrot_image(x_0, y_0, zoom, &set);
118         }
119         if is_key_down(KeyCode::E) {
120             zoom -= zoom_mult * zoom;
121             screen = mandelbrot_image(x_0, y_0, zoom, &set);
122         }
123         if is_key_down(KeyCode::Q) {
124             zoom += zoom_mult * zoom;
125             screen = mandelbrot_image(x_0, y_0, zoom, &set);
126         }
127         if is_key_down(KeyCode::R) {
128             set.max_iter += 25;
129             screen = mandelbrot_image(x_0, y_0, zoom, &set);
130         }
131         if is_key_down(KeyCode::F) {
132             set.max_iter -= 25;
133             screen = mandelbrot_image(x_0, y_0, zoom, &set);
134         }
135         if is_mouse_button_down(MouseButton::Left) {
136             let mouse = mouse_position();
137             let ox = screen_width() / 2.0;
138             let oy = screen_height() / 2.0;
139             let unit;
140             if ox < oy {
141                 unit = ((zoom * 2.) / screen_width()).abs();
142             } else {
143                 unit = ((zoom * 2.) / screen_height()).abs();
144             };
145             x_0 = (mouse.0 - ox) * unit + x_0;
146             y_0 = (mouse.1 - oy) * unit + y_0;
147             screen = mandelbrot_image(x_0, y_0, zoom, &set);
148         }
149 
150         let texture = Texture2D::from_image(&screen);
151         draw_texture(texture, 0.0, 0.0, WHITE);
152 
153         if is_key_down(KeyCode::Enter) {
154             let image = get_screen_data();
155             image.export_png(&format!(
156                 "__out/mandelbrot/mandelbrot_{}.png",
157                 rng.gen::<i32>()
158             ));
159             return;
160         }
161         next_frame().await
162     }
163 }
