 18 #[derive(Clone, Debug)]
 19 struct Point<T> {
 20     x: T,
 21     y: T,
 22 }
 23 
 24 impl Point<f64> {
 25     fn to_world(&self, singl: &Singleton) -> Point<f64> {
 26         let unit = map_screen_to_world(&singl);
 27         Point::<f64> {
 28             x: (self.x - singl.offset.1.x as f64 - screen_width() as f64 / 2f64) * unit
 29                 + singl.center.x,
 30             y: -(self.y - singl.offset.1.y as f64 - screen_height() as f64 / 2f64) * unit
 31                 + singl.center.y,
 32         }
 33     }
 34 }

 81 impl Singleton {
 82     fn generate_colors(&mut self) {
 83         let colorscheme = colorschemes::colorschemes()[self.colorscheme].clone();
 84         self.pallet = Vec::new();
 85         let color = self.max_iter / (colorscheme.len() - 1);
 86         for i in 0..(colorscheme.len() - 1) {
 87             for j in 0..color {
 88                 self.pallet.push(Color::new(
 89                     colorscheme[i].r
 90                         + (colorscheme[i + 1].r - colorscheme[i].r) * (j as f32 / color as f32),
 91                     colorscheme[i].g
 92                         + (colorscheme[i + 1].g - colorscheme[i].g) * (j as f32 / color as f32),
 93                     colorscheme[i].b
 94                         + (colorscheme[i + 1].b - colorscheme[i].b) * (j as f32 / color as f32),
 95                     colorscheme[i].a
 96                         + (colorscheme[i + 1].b - colorscheme[i].b) * (j as f32 / color as f32),
 97                 ));
 98             }
 99         }
100         while self.pallet.len() <= self.max_iter {
101             self.pallet.push(BLACK);
102         }
103     }
104 }

106 fn mandelbrot(c: num::complex::Complex<f64>, singl: &Singleton) -> usize {
107     let mut z = num::complex::Complex::<f64>::new(singl.julia.x, singl.julia.y);
108     let mut i: usize = 0;
109     while i < singl.max_iter && z.l1_norm() <= 4f64 {
110         z = z.powf(singl.power) + c;
111         i += 1;
112     }
113     return i;
114 }
115 fn map_screen_to_world(singl: &Singleton) -> f64 {
116     let world_unit: f64;
117     if screen_width() < screen_height() {
118         world_unit = 4f64 / (screen_width() as f64 * singl.scale);
119     } else {
120         world_unit = 4f64 / (screen_height() as f64 * singl.scale);
121     }
122     return world_unit;
123 }

125 fn fractal(singl: &Singleton) -> Vec<Texture2D> {
126     let mut bands = Vec::new();
127     for i in 0..singl.bands {
128         bands.push(i);
129     }
130 
131     let mut images = Vec::new();
132     let mut band_height = screen_height() as usize / singl.bands;
133     band_height += band_height * singl.bands / 10;
134     for _ in 0..singl.bands {
135         images.push(Image::gen_image_color(
136             screen_width() as u16,
137             band_height as u16,
138             WHITE,
139         ));
140     }
141 
142     let bands_mutex = Arc::new(Mutex::new(bands));
143     let images_mutex = Arc::new(Mutex::new(images));
144     let singl_mutex = Arc::new(singl.clone());
145 
146     let mut handles = Vec::new();
147     for _ in 0..singl.threads {
148         let singl_clone = Arc::clone(&singl_mutex);
149         let bands_clone = Arc::clone(&bands_mutex);
150         let images_clone = Arc::clone(&images_mutex);
151         let handle = thread::spawn(move || {
152             let local_singl = singl_clone;
153             loop {
154                 let mut bands = bands_clone.lock().unwrap();
155                 if bands.len() == 0 {
156                     break;
157                 }
158                 let index = bands.remove(0);
159                 drop(bands);
160 
161                 let images = images_clone.lock().unwrap();
162                 let width = images[index].width();
163                 let height = images[index].height();
164                 drop(images);
165 
166                 let mut fractal = Image::gen_image_color(width as u16, height as u16, WHITE);
167 
168                 for x in 0..screen_width() as u32 {
169                     for y in 0..height as u32 {
170                         let point = Point::<f64> {
171                             x: x as f64,
172                             y: (index * height) as f64 + y as f64,
173                         }
174                         .to_world(&local_singl);
175                         let c = num::complex::Complex::<f64>::new(point.x, point.y);
176 
177                         let iter = mandelbrot(c, &local_singl);
178 
179                         fractal.set_pixel(x, y, local_singl.pallet[iter]);
180                     }
181                 }
182 
183                 let mut images = images_clone.lock().unwrap();
184                 images[index] = fractal;
185                 drop(images);
186                 thread::sleep(Duration::from_millis(1));
187             }
188         });
189         handles.push(handle);
190     }
191 
192     for h in handles {
193         h.join().unwrap();
194     }
195 
196     let mut textures = Vec::new();
197     let images_clone = Arc::clone(&images_mutex);
198     let images = images_clone.lock().unwrap();
199     for i in 0..singl.bands {
200         textures.push(Texture2D::from_image(&images[i]));
201     }
202     drop(images);
203     return textures;
204 }

206 fn draw_menus(singl: &mut Singleton) {
207     if singl.egui {
208         egui_macroquad::ui(|egui_ctx| {
209             egui::Window::new("Settings").show(egui_ctx, |ui| {
210                 ui.add(egui::Slider::new(&mut singl.scale, 1f64..=1_000_000f64).text("Zoom"));
211                 ui.add(egui::Slider::new(&mut singl.max_iter, 0..=1_000).text("Max iterations"));
212                 ui.add(egui::Slider::new(&mut singl.power, 0.0..=100.0).text("Power"));
213                 ui.add(egui::Slider::new(&mut singl.threads, 1..=16).text("Threads"));
214                 ui.add(
215                     egui::Slider::new(&mut singl.animation_unit, 0.0001..=0.1)
216                         .text("Animation unit"),
217                 );
218                 ui.add(
219                     egui::Slider::new(&mut singl.refresh_limit, 10..=10000).text("Redraw delay"),
220                 );
221 
222                 if ui.button("Refresh").clicked() {
223                     singl.refresh = true;
224                     singl.mouse_click = false;
225                     singl.offset = (Point { x: 0., y: 0. }, Point { x: 0., y: 0. });
226                 }
227                 if ui.button("Animation on/off").clicked() {
228                     singl.animation = !singl.animation;
229                 }
230             });
231 
232             egui::Window::new("Debugg info").show(egui_ctx, |ui| {
233                 ui.label(format!("Scale: {}", singl.scale));
234                 ui.label(format!("Iterations: {}", singl.max_iter));
235                 ui.label(format!("Refresh: {}", singl.refresh));
236                 ui.label(format!("Mouse click: {}", singl.mouse_click));
237 
238                 ui.collapsing("Positions", |ui| {
239                     ui.label(format!("Center: ({}, {})", singl.center.x, singl.center.y));
240                     ui.label(format!(
241                         "Offset: ({}, {}), ({}, {})",
242                         singl.offset.0.x, singl.offset.0.y, singl.offset.1.x, singl.offset.1.y
243                     ));
244                     ui.label(format!("Mouse position: {:?}", mouse_position()));
245                     ui.label(format!(
246                         "World position: {:?}",
247                         Point::<f64> {
248                             x: mouse_position().0 as f64,
249                             y: mouse_position().1 as f64
250                         }
251                         .to_world(&singl)
252                     ));
253                 });
254 
255                 ui.collapsing("Colors", |ui| {
256                     ui.monospace(format!("{:#?}", singl.pallet.len()));
257                 });
258 
259                 if ui.button("Reset").clicked() {
260                     *singl = Singleton {
261                         ..Default::default()
262                     };
263                 }
264                 if ui.button("Center").clicked() {
265                     singl.center = Point::<f64> { x: 0., y: 0. };
266                     singl.offset = (Point { x: 0., y: 0. }, Point { x: 0., y: 0. });
267                     singl.mouse_click = false;
268                 }
269             });
270         });
271         egui_macroquad::draw();
272     }
273 }

275 fn user_input(singl: &mut Singleton) {
276     if singl.refresh {
277         return;
278     }
279     let xrest = 300.;
280     let yrest = 400.;
281     if singl.egui {
282         draw_rectangle(0., 0., xrest, yrest, Color::new(0., 0., 0., 0.2));
283     }
284     if mouse_position().0 > xrest || mouse_position().1 > yrest || !singl.egui {
285         if is_mouse_button_pressed(MouseButton::Right) {
286             let mouse = mouse_position();
287             singl.julia = Point::<f64> {
288                 x: mouse.0 as f64,
289                 y: mouse.1 as f64,
290             }
291             .to_world(&singl);
292             singl.refresh = true;
293         }
294         if is_mouse_button_pressed(MouseButton::Left) && !singl.mouse_click {
295             let mouse = mouse_position();
296             singl.center = Point::<f64> {
297                 x: mouse.0 as f64,
298                 y: mouse.1 as f64,
299             }
300             .to_world(&singl);
301             singl.refresh = true;
302         }
303 
304         if mouse_wheel().1 != 0. {
305             singl.scale += singl.scale * (mouse_wheel().1 / 10.) as f64;
306             // singl.refresh = true;
307             singl.last_refresh = Instant::now();
308         }
309     }
310 
311     if is_key_pressed(KeyCode::Enter) {
312         singl.refresh = true;
313     }
314 
315     if is_key_pressed(KeyCode::Escape) {
316         singl.egui = !singl.egui;
317     }
318 
319     if is_key_pressed(KeyCode::Space) {
320         singl.animation = !singl.animation;
321     }
322 
323     if is_key_pressed(KeyCode::Tab) {
324         singl.colorscheme += 1usize;
325         if singl.colorscheme >= colorschemes::colorschemes().len() {
326             singl.colorscheme = 0usize;
327         }
328         singl.generate_colors();
329         singl.refresh = true;
330     }
331 }

333 #[macroquad::main(window_conf)]
334 async fn main() {
335     let mut singl = Singleton {
336         power: 2.,
337         egui: false,
338         ..Default::default()
339     };
340     singl.generate_colors();
341 
342     let mut textures = fractal(&singl);
343 
344     loop {
345         clear_background(LIGHTGRAY);
346 
347         if singl.last_refresh.elapsed().as_millis() > singl.refresh_limit as u128 {
348             singl.refresh = true;
349         }
350 
351         if singl.refresh {
352             if singl.pallet.len() < singl.max_iter {
353                 singl.generate_colors();
354             }
355 
356             textures = fractal(&singl);
357 
358             singl.last_refresh = Instant::now();
359             singl.refresh = false;
360         }
361 
362         if singl.animation {
363             singl.power += singl.animation_unit;
364         }
365 
366         for i in 0..singl.bands {
367             draw_texture(textures[i], 0., i as f32 * textures[i].height(), WHITE);
368         }
369 
370         user_input(&mut singl);
371         draw_menus(&mut singl);
372 
373         window::next_frame().await
374     }
375 }
