 29 #define PI 3.14159265
 30 POINT rotate_point(POINT c, POINT p, FLOAT angle)
 31 {
 32   angle =  angle * PI / 180;
 33   float _sin = sin(angle);
 34   float _cos = cos(angle);
 35   p.x -= c.x;
 36   p.y -= c.y;
 37   float xnew = p.x * _cos - p.y * _sin;
 38   float ynew = p.x * _sin + p.y * _cos;
 39   p.x = xnew + c.x;
 40   p.y = ynew + c.y;
 41   return p;
 42 }

 44 POINT center_apt() {
 45     POINT center;
 46     center.x = (apt[0].x + apt[3].x) / 2.;
 47     center.y = (apt[0].y + apt[3].y) / 2.;
 48     return center;
 49 }

169

348             {

379     }
