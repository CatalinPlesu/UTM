  8 static TCHAR szWindowClass[] = _T("DesktopApp");
  9 static TCHAR szTitle[] = _T("Windows Desktop Guided Tour Application");
 10 
 11 HINSTANCE hInst;
 12 
 13 LRESULT CALLBACK WndProc(HWND, UINT, WPARAM, LPARAM);
 14 
 15 int WINAPI WinMain(
 16    _In_ HINSTANCE hInstance,
 17    _In_opt_ HINSTANCE hPrevInstance,
 18    _In_ LPSTR     lpCmdLine,
 19    _In_ int       nCmdShow
 20 )
 21 {
 22    WNDCLASSEX wcex;
 23 
 24    wcex.cbSize = sizeof(WNDCLASSEX);
 25    wcex.style          = CS_HREDRAW | CS_VREDRAW;
 26    wcex.lpfnWndProc    = WndProc;
 27    wcex.cbClsExtra     = 0;
 28    wcex.cbWndExtra     = 0;
 29    wcex.hInstance      = hInstance;
 30    wcex.hIcon          = LoadIcon(wcex.hInstance, IDI_APPLICATION);
 31    wcex.hCursor        = LoadCursor(NULL, IDC_ARROW);
 32    wcex.hbrBackground  = (HBRUSH)(COLOR_WINDOW+1);
 33    wcex.lpszMenuName   = NULL;
 34    wcex.lpszClassName  = szWindowClass;
 35    wcex.hIconSm        = LoadIcon(wcex.hInstance, IDI_APPLICATION);
 36 
 37 
 38    if (!RegisterClassEx(&wcex))
 39    {
 40       MessageBox(NULL,
 41          _T("Call to RegisterClassEx failed!"),
 42          _T("Windows Desktop Guided Tour"),
 43          MB_OK);
 44 
 45       return 1;
 46    }
 47 
 48    hInst = hInstance;
 49 
 50    HWND hWnd = CreateWindowEx(
 51       WS_EX_OVERLAPPEDWINDOW,
 52       szWindowClass,
 53       szTitle,
 54       WS_OVERLAPPEDWINDOW,
 55       CW_USEDEFAULT, CW_USEDEFAULT,
 56       500, 500,
 57       NULL,
 58       NULL,
 59       hInstance,
 60       NULL
 61    );
 62 
 63    if (!hWnd)
 64    {
 65       MessageBox(NULL,
 66          _T("Call to CreateWindow failed!"),
 67          _T("Windows Desktop Guided Tour"),
 68          MB_OK);
 69 
 70       return 1;
 71    }
 72 
 73    ShowWindow(hWnd,
 74       nCmdShow);
 75    UpdateWindow(hWnd);
 76 
 77    MSG msg;
 78    while (GetMessage(&msg, NULL, 0, 0))
 79    {
 80       TranslateMessage(&msg);
 81       DispatchMessage(&msg);
 82    }
 83 
 84    return (int) msg.wParam;
 85 }
 86 
 87 LRESULT CALLBACK WndProc(HWND hWnd, UINT message, WPARAM wParam, LPARAM lParam)
 88 {
 89    PAINTSTRUCT ps;
 90    HDC hdc;
 91    TCHAR greeting[] = _T("This Program Was Compiled On Gnu+Linux");
 92 
 93    switch (message)
 94    {
 95    case WM_PAINT:
 96       hdc = BeginPaint(hWnd, &ps);
 97 
 98      RECT rect;
 99      GetClientRect(hWnd, &rect); //Obţinerea suprafeţei de desenare 
100      DrawText(hdc, greeting, -1, &rect, DT_SINGLELINE| DT_CENTER | DT_VCENTER);
101 
102      // random color
103 
104       EndPaint(hWnd, &ps);
105       break;
106    case WM_DESTROY:
107       PostQuitMessage(0);
108       break;
109    default:
110       return DefWindowProc(hWnd, message, wParam, lParam);
111       break;
112    }
113 
114    return 0;
115 }
