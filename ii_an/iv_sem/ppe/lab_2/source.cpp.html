 32 point rotate_point(point c, point p, float angle)
 33 {
 34   angle =  angle * PI / 180;
 35   float _sin = sin(angle);
 36   float _cos = cos(angle);
 37   p.x -= c.x;
 38   p.y -= c.y;
 39   float xnew = p.x * _cos - p.y * _sin;
 40   float ynew = p.x * _sin + p.y * _cos;
 41   p.x = xnew + c.x;
 42   p.y = ynew + c.y;
 43   return p;
 44 }

 46 line rotate_line(line l, float angle) {
 47 
 48     l.a = rotate_point(l.c, l.a, angle);
 49     l.b = rotate_point(l.c, l.b, angle);
 50     return l;
 51 }

197 LRESULT CALLBACK WndProc(HWND hWnd, UINT message,
198     WPARAM wParam, LPARAM lParam)
199 {
200 
201     switch (message)
202     {
203     case WM_TIMER:
204         RedrawWindow(hWnd,NULL,NULL,RDW_INVALIDATE | RDW_ERASE);

208         return 0;

210         PostQuitMessage(0);
211         return 0;
212     default:

214     }
