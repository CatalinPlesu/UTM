 20 int CALLBACK WinMain(_In_ HINSTANCE hInstance, _In_opt_ HINSTANCE hPrevInstance,
 21     _In_ LPSTR lpCmdLine, _In_ int nCmdShow)
 22 {
 23     WNDCLASSEX wcex;
 24     wcex.cbSize = sizeof(WNDCLASSEX);
 25     wcex.style = CS_HREDRAW | CS_VREDRAW;
 26     wcex.lpfnWndProc = WndProc;
 27     wcex.cbClsExtra = 0;
 28     wcex.cbWndExtra = 0;
 29     wcex.hInstance = hInstance;
 30     wcex.hCursor = LoadCursor(NULL, IDC_ARROW);
 31     wcex.hIcon = LoadIcon(NULL, IDI_APPLICATION);
 32     wcex.hbrBackground = (HBRUSH)(COLOR_WINDOW + 1);
 33     wcex.lpszMenuName = NULL;
 34     wcex.lpszClassName = szWindowClass;
 35     wcex.hIconSm = LoadIcon(NULL, IDI_APPLICATION);
 36 
 37     if (!RegisterClassEx(&wcex)) {
 38         MessageBox(NULL, _T("Call to RegisterClassEx failed!"),
 39             _T("Windows Desktop Guided Tour"), NULL);
 40         return 1;
 41     }
 42 
 43     hInst = hInstance;
 44     HWND hwnd = CreateWindow(szWindowClass, szTitle, WS_OVERLAPPEDWINDOW | WS_VSCROLL,
 45         0, 0, 500, 550, NULL, NULL, hInstance, NULL);
 46 
 47     if (!hwnd) {
 48         MessageBox(NULL, _T("Call to CreateWindow failed!"),
 49             _T("Windows Desktop Guided Tour"), NULL);
 50         return 1;
 51     }
 52 
 53     ShowWindow(hwnd, nCmdShow);
 54     UpdateWindow(hwnd);
 55 
 56     MSG msg;
 57     while (GetMessage(&msg, NULL, 0, 0)) {
 58         TranslateMessage(&msg);
 59         DispatchMessage(&msg);
 60     }
 61 
 62     return (int)msg.wParam;
 63 }

 65 #define winmacro(text, style) CreateWindow(TEXT("button"), TEXT(text), \
 66             WS_CHILD | WS_VISIBLE |  ES_LEFT | style, \
 67             10, y+=20, 400, 20, \
 68             hwnd, NULL, ((LPCREATESTRUCT)lParam)->hInstance, NULL)

 70 LRESULT CALLBACK WndProc(HWND hwnd, UINT message, WPARAM wParam,
 71     LPARAM lParam)
 72 {
 73     PAINTSTRUCT ps;
 74     SCROLLINFO si;
 75     HDC hdc;
 76 
 77     switch (message) {
 78     case WM_TIMER:
 79         KillTimer(hwnd, TIMER);
 80 
 81         MessageBox(NULL,
 82             TEXT("Timpul a expirat!"),
 83             TEXT("Alarma"),
 84             MB_ICONEXCLAMATION);
 85 
 86         PostQuitMessage(0);
 87         break;
 88     case WM_CREATE: {
 89         MessageBox(NULL,
 90             TEXT("Vei avea la dispozitie 2 minute"),
 91             TEXT("Atentie"),
 92             MB_ICONINFORMATION);
 93 
 94         SetTimer(hwnd, TIMER, TOTAL_TIME, NULL);
 95 
 96         int y = 10;
 97         winmacro("Cocoa", WS_GROUP | BS_AUTORADIOBUTTON);
 98         winmacro("WinApi",  BS_AUTORADIOBUTTON );
 99         winmacro("Xlib",  BS_AUTORADIOBUTTON );
100         winmacro("wayland-client",  BS_AUTORADIOBUTTON );
101 
102         y = 20*6+10;
103         winmacro("NT",  WS_GROUP | BS_AUTORADIOBUTTON );
104         winmacro("Linux",  BS_AUTORADIOBUTTON );
105         winmacro("XNU",  BS_AUTORADIOBUTTON );
106 
107         y = 20*12+10;
108         winmacro("GIMP",   BS_AUTOCHECKBOX);
109         winmacro("Photoshop",   BS_AUTOCHECKBOX);
110         winmacro("Emacs",   BS_AUTOCHECKBOX);
111         winmacro("Cat",   BS_AUTOCHECKBOX);
112         winmacro("Vim",   BS_AUTOCHECKBOX);
113         winmacro("Bash",   BS_AUTOCHECKBOX);
114         winmacro("Powershell",   BS_AUTOCHECKBOX);
115         winmacro("Zsh",   BS_AUTOCHECKBOX);
116         winmacro("gcc",   BS_AUTOCHECKBOX);
117         winmacro("wget",   BS_AUTOCHECKBOX);
118 
119 
120         CreateWindow(
121             "BUTTON", "Finiseaza", WS_VISIBLE | WS_CHILD | WS_BORDER, 10,
122             500, 90, 20, hwnd, (HMENU)PUSH_BUTTON,
123             ((LPCREATESTRUCT)lParam)->hInstance, NULL);
124         break;
125     }
126 
127     case WM_COMMAND: {
128         if (wParam == PUSH_BUTTON) {
129 
130         MessageBox(NULL,
131             TEXT("Testul a fost finisat"),
132             TEXT("Felicitari"),
133             MB_ICONINFORMATION);
134 
135         PostQuitMessage(0);
136         }
137         break;
138     }
139 
140     case WM_PAINT: {
141         hdc = BeginPaint(hwnd, &ps);
142 
143         TCHAR* intrebari[] = {
144             _T("Cel mai rau API pentru interfata de utilizator din univers"),
145             _T("Cel mai bun kernel"),
146             _T("Programe GNU"),
147         };
148         TextOut(hdc, 10, 10, intrebari[0], _tcslen(intrebari[0]));
149         TextOut(hdc, 10, 10 + 20*6, intrebari[1], _tcslen(intrebari[1]));
150         TextOut(hdc, 10, 10 + 20*12, intrebari[2], _tcslen(intrebari[2]));
151 
152         EndPaint(hwnd, &ps);
153         DeleteDC(hdc);
154         break;
155     }
156 
157     case WM_SIZE:
158         yClient = HIWORD(lParam);
159 
160         SetScrollRange(hwnd, SB_VERT, 0, total_scroll_points, TRUE);
161         InvalidateRect(hwnd, NULL, TRUE);
162         break;
163 
164     case WM_VSCROLL: {
165         si.cbSize = sizeof(si);
166         si.fMask = SIF_ALL;
167         GetScrollInfo(hwnd, SB_VERT, &si);
168 
169 
170         switch (LOWORD(wParam)) {
171         case SB_TOP:
172             si.nPos = si.nMin;
173             break;
174         case SB_BOTTOM:
175             si.nPos = si.nMax;
176             break;
177         case SB_LINEUP:
178             si.nPos -= 1;
179             break;
180         case SB_LINEDOWN:
181             si.nPos += 1;
182             break;
183         case SB_THUMBTRACK:
184             si.nPos = si.nTrackPos;
185             break;
186         default:
187             break;
188         }
189         // Set the new scollbar position.
190         si.fMask = SIF_POS;
191         SetScrollInfo(hwnd, SB_VERT, &si, TRUE);
192         // redraw window contents by manually sending manually sending WM_PAINT
193         InvalidateRect(hwnd, NULL, TRUE);
194     } break;
195 
196     case WM_DESTROY: {
197         PostQuitMessage(0);
198         break;
199     }
200     default:
201         return DefWindowProc(hwnd, message, wParam, lParam);
202         break;
203     }
204     return 0;
205 }
